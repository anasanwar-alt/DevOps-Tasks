TASK 1 - network segmentation

$ docker network inspect day6_frontend_net --format 'frontend_net: {{range .Containers}}{{.Name}} {{end}}'
frontend_net: day6-backend-1 day6-frontend-1 

$ docker network inspect day6_backend_net  --format 'backend_net : {{range .Containers}}{{.Name}} {{end}}'
backend_net : day6-db-1 day6-backend-1 

# day6-db-1 and day6-frontend-1 never appear on the same network

$ docker exec day6-frontend-1 wget -T 3 -qO- http://db:5432
wget: bad address 'db:5432'
# 'bad address' is a DNS failure - it happens before any packet is sent.
# Docker's embedded DNS only answers for containers sharing a network.

$ docker exec day6-frontend-1 wget -T 3 -qO- http://backend:3000/health
ok
# ...but the backend still answers. Segmented, not broken - which is
# what distinguishes this from simply having misconfigured the network.
